cmake_minimum_required(VERSION 3.24)
project(SN2MCPNative LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Subnautica 2 MCP requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SN2_MCP_DEVELOPMENT_PROBES "Build isolated game-Tick acceptance fixtures (excluded from production)" OFF)
option(SN2_MCP_MOD_SDK_ONLY "Build the standalone native modification SDK only" OFF)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
include(FetchContent)
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_Declare(json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
FetchContent_Declare(minhook
    GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
    GIT_TAG 8af6b4acae5a9388fd742b56fa79ece89d96f823)
FetchContent_MakeAvailable(minhook)

set(SN2_GAME_ACCESS_SOURCES
    src/game_access.cpp src/game_access_array.cpp src/game_access_containers.cpp
    src/game_access_struct.cpp src/game_access_vector.cpp src/game_access_transform.cpp
    src/game_access_object_array.cpp src/game_access_linear_color.cpp)
add_library(sn2_mod_runtime OBJECT src/mod_entry.cpp src/observations.cpp ${SN2_GAME_ACCESS_SOURCES})
target_include_directories(sn2_mod_runtime PUBLIC include)
target_link_libraries(sn2_mod_runtime PUBLIC nlohmann_json::nlohmann_json minhook bcrypt)
target_compile_options(sn2_mod_runtime PRIVATE /utf-8 /W4 /EHsc)
target_compile_definitions(sn2_mod_runtime PRIVATE SN2_MOD_RUNTIME=1)
if(SN2_MCP_MOD_SDK_ONLY)
    return()
endif()

add_library(sn2_mcp_transport STATIC src/transport.cpp)
target_include_directories(sn2_mcp_transport PUBLIC include)
target_link_libraries(sn2_mcp_transport PUBLIC nlohmann_json::nlohmann_json ws2_32)
target_compile_options(sn2_mcp_transport PRIVATE /utf-8 /W4)
add_library(sn2_mcp_observations STATIC src/observations.cpp)
target_include_directories(sn2_mcp_observations PUBLIC include)
target_link_libraries(sn2_mcp_observations PUBLIC sn2_mcp_transport)
target_compile_options(sn2_mcp_observations PRIVATE /utf-8 /W4 /EHsc)

add_library(sn2_mcp_native SHARED src/entry.cpp ${SN2_GAME_ACCESS_SOURCES})
target_include_directories(sn2_mcp_native PRIVATE include)
target_link_libraries(sn2_mcp_native PRIVATE sn2_mcp_transport sn2_mcp_observations minhook bcrypt)
target_compile_options(sn2_mcp_native PRIVATE /utf-8 /W4 /EHsc)
if(SN2_MCP_DEVELOPMENT_PROBES)
    target_compile_definitions(sn2_mcp_native PRIVATE SN2_MCP_DEVELOPMENT_PROBES=1)
endif()
set_target_properties(sn2_mcp_native PROPERTIES OUTPUT_NAME "SN2_MCP")
